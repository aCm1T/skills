\topic{L1: Definitions}{Example}
\method{Expectation}
For discrete $X$ with mass function $p$, if the sum is absolutely convergent,
\[
  \E[g(X)]=\sum_x g(x)p(x).
\]
For a continuous density, replace the sum by an integral.
\method{Variance}
If $\E[X^2]<\infty$, writing $\mu=\E[X]$ gives
\[
\begin{aligned}
 \Var(X)&=\E[(X-\mu)^2]\\
        &=\E[X^2]-\mu^2.
\end{aligned}
\]
\caution{State assumptions and parameter conventions beside the formula.}
\columnbreak
\topic{L2: Methods}{Example}
\method{Normal mean interval}
For independent $X_i\sim N(\mu,\sigma^2)$ with known $\sigma>0$,
\[
 Z=\frac{\overline X-\mu}{\sigma/\sqrt n}\sim N(0,1).
\]
Let $z_q=\Phi^{-1}(q)$. A two-sided interval of coverage $1-\alpha$ is
\[
 \overline X\ \pm\ z_{1-\alpha/2}\frac{\sigma}{\sqrt n}.
\]
\caution{Unknown variance needs a different pivot; distinguish exact from asymptotic results.}
\columnbreak
\topic{L3: Application}{Example}
\method{Using an interval}
Identify the target, verify the assumptions, choose the pivot and quantile, then substitute.
For $n=25$, $\overline x=10$ and known $\sigma=2$, the 95\% interval is
\[
 10\pm 1.96\frac{2}{5}=[9.216,10.784].
\]

\topic{L4: Transformation}{Example}
\method{Inverse CDF}
For a continuous strictly increasing CDF $F$ and $U\sim\mathrm{Uniform}(0,1)$,
\[
 U=F(X)\quad\Longrightarrow\quad X=F^{-1}(U).
\]
For $F(x)=x^r$ on $0\le x\le1$, $r>0$,
\[
 U=X^r\quad\Longrightarrow\quad X=U^{1/r}.
\]
For a general CDF use its generalized inverse.
\caution{Retain the intermediate relation that determines the direction of the transform.}
\columnbreak
\topic{L5: Dependence}{Example}
\method{Covariance matrix}
With finite second moments,
\[
 \Sigma=\begin{pmatrix}
 \Var(X)&\Cov(X,Y)\\
 \Cov(X,Y)&\Var(Y)
 \end{pmatrix}.
\]
For a fixed vector $a$ and random vector $V$ with covariance $\Sigma$,
\[
 \Var(a^{\mathsf T}V)=a^{\mathsf T}\Sigma a.
\]
\caution{Check matrix dimensions; zero covariance alone does not imply independence.}
